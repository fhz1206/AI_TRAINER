# 3. 分词/反分词函数（必须和训练时的TextDataset分词逻辑完全一致）
def tokenize(text, vocab_size, pad_token_id=0):
    """使用训练时的显式词表进行分词，和TextDataset的char2token映射一致"""
    if char2token is not None:
        # 使用训练时保存的显式词表映射（和训练完全一致）
        unknown_idx = list(char2token.values())[0] if char2token else 0
        tokens = [char2token.get(char, unknown_idx) for char in text]
    else:
        # 兜底：无词表时按字分词，用排序后的字符集映射（和TextDataset初始化逻辑一致）
        char_set = sorted(set(text))
        fallback_vocab = {char: idx for idx, char in enumerate(char_set)}
        unknown_idx = 0
        tokens = [fallback_vocab.get(char, unknown_idx) for char in text]
    return torch.tensor(tokens, dtype=torch.long, device='cpu')

def detokenize(token_ids, vocab_size):
    """使用训练时的显式词表将token ID还原为文字"""
    if isinstance(token_ids, torch.Tensor):
        token_ids = token_ids.tolist()
    if token2char is not None:
        # 使用训练时保存的显式词表反向映射，还原为可读文字
        chars = []
        for tok in token_ids:
            if tok in token2char:
                chars.append(token2char[tok])
            # 跳过pad_token和-100等特殊标记
        return "".join(chars)
    else:
        # 无词表兜底：输出数字标记
        return " ".join([f"<tok_{tok}>" for tok in token_ids if tok != 0])
