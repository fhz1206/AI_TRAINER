# 手部关键点检测（官方Tasks API标准写法，上传图片直接运行）
import cv2
import numpy as np
import base64
import mediapipe as mp

# -------------------------- 初始化内置模型（自动加载，无需修改） --------------------------
print("正在加载内置手部检测模型...")
try:
    if '_global_hand_models' not in globals() or _global_hand_models is None:
        _global_hand_models = _init_hand_models()
    detector = _global_hand_models
    if detector is None:
        raise RuntimeError("模型初始化返回空值")
    print("✅ 模型加载完成")
except Exception as e:
    raise RuntimeError(f"模型加载失败：{str(e)}")

# -------------------------- 手部检测逻辑（官方Tasks API标准调用） --------------------------
img_path = test_data_path
print(f"正在读取测试图片：{img_path}")
img = cv2.imread(img_path)
if img is None:
    raise ValueError(f"无法读取图片：{img_path}，请检查文件是否损坏")
img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
h, w = img.shape[:2]

# 执行手部检测
print("正在检测手部关键点...")
if detector is None:
    raise RuntimeError("手部检测模型未成功加载，请检查后端环境")
mp_image = mp.Image(image_format=mp.ImageFormat.SRGB, data=img_rgb)
results = detector.detect(mp_image)

# 处理检测结果
if not results.hand_landmarks:
    print("❌ 未检测到手部，请上传清晰的手部照片（正面、背景简洁、手部无遮挡）")
    _hand_detection_result = {
        'error': '未检测到手部',
        'message': '未检测到手部，请上传清晰的手部照片（正面、背景简洁、手部无遮挡）'
    }
else:
    # 提取21个关键点，转像素坐标
    hand_landmarks = results.hand_landmarks[0]
    keypoints = []
    for lm in hand_landmarks:
        x = int(lm.x * w)
        y = int(lm.y * h)
        conf = 1.0
        keypoints.append([x, y, conf])
    keypoints = np.array(keypoints, dtype=np.float32)

    # 关键点名称
    keypoint_names = ["手腕", "大拇指根部", "大拇指第一关节", "大拇指第二关节", "大拇指指尖",
                      "食指根部", "食指第一关节", "食指第二关节", "食指指尖",
                      "中指根部", "中指第一关节", "中指第二关节", "中指指尖",
                      "无名指根部", "无名指第一关节", "无名指第二关节", "无名指指尖",
                      "小指根部", "小指第一关节", "小指第二关节", "小指指尖"]

    print("\n=== 手部关键点检测结果 ===")
    keypoint_list = []
    for idx, (x, y, conf) in enumerate(keypoints):
        print(f"关键点{idx}（{keypoint_names[idx]}）：x={int(x)}, y={int(y)}, 置信度={float(conf):.4f}")
        keypoint_list.append({
            'id': idx,
            'name': keypoint_names[idx],
            'x': int(x),
            'y': int(y),
            # 🛠️ 修复：转成原生float，确保JSON能序列化
            'confidence': float(conf)
        })

    # 生成基础标注图
    basic_img = img_rgb.copy()
    for idx, (x, y, conf) in enumerate(keypoints):
        if conf > 0.5:
            cv2.circle(basic_img, (int(x), int(y)), 5, (0, 255, 0), -1)
            cv2.putText(basic_img, str(idx), (int(x)+10, int(y)-10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0,0,255), 1)
    _, basic_buffer = cv2.imencode('.jpg', cv2.cvtColor(basic_img, cv2.COLOR_RGB2BGR))
    basic_base64 = base64.b64encode(basic_buffer).decode('utf-8')

    # 生成彩色骨架图
    HAND_CONNECTIONS = [
        (0,1),(1,2),(2,3),(3,4), (0,5),(5,6),(6,7),(7,8),
        (0,9),(9,10),(10,11),(11,12), (0,13),(13,14),(14,15),(15,16),
        (0,17),(17,18),(18,19),(19,20), (5,9),(9,13),(13,17)
    ]
    CONNECTION_COLORS = {
        (0,1):(255,165,0), (1,2):(255,165,0), (2,3):(255,165,0), (3,4):(255,165,0),
        (0,5):(0,255,0), (5,6):(0,255,0), (6,7):(0,255,0), (7,8):(0,255,0),
        (0,9):(0,0,255), (9,10):(0,0,255), (10,11):(0,0,255), (11,12):(0,0,255),
        (0,13):(128,0,128), (13,14):(128,0,128), (14,15):(128,0,128), (15,16):(128,0,128),
        (0,17):(255,0,0), (17,18):(255,0,0), (18,19):(255,0,0), (19,20):(255,0,0),
        (5,9):(255,255,0), (9,13):(255,255,0), (13,17):(255,255,0)
    }
    skeleton_img = img_rgb.copy()
    for conn in HAND_CONNECTIONS:
        s_idx, e_idx = conn
        s_pt = keypoints[s_idx]
        e_pt = keypoints[e_idx]
        if s_pt[2] > 0.5 and e_pt[2] > 0.5:
            color = CONNECTION_COLORS.get(conn, (255,255,255))
            cv2.line(skeleton_img, (int(s_pt[0]), int(s_pt[1])), (int(e_pt[0]), int(e_pt[1])), color, 3)
    for idx, (x, y, conf) in enumerate(keypoints):
        if conf > 0.5:
            cv2.circle(skeleton_img, (int(x), int(y)), 6, (0,255,255), -1)
            cv2.putText(skeleton_img, str(idx), (int(x)+10, int(y)-10), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255,0,0), 2)
    _, skeleton_buffer = cv2.imencode('.jpg', cv2.cvtColor(skeleton_img, cv2.COLOR_RGB2BGR))
    skeleton_base64 = base64.b64encode(skeleton_buffer).decode('utf-8')

    _hand_detection_result = {
        'keypoints': keypoint_list,
        'basic_image': f"data:image/jpeg;base64,{basic_base64}",
        'skeleton_image': f"data:image/jpeg;base64,{skeleton_base64}",
        'message': '检测成功'
    }
    print(f"\n✅ 检测完成，共识别到{len(keypoint_list)}个关键点")
