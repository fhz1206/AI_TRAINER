# 扩散图像编辑模型测试模板（SDEdit 式：源图加噪后条件化去噪修复）
import os
import cv2
import numpy as np
import torch
from model_io import load_model

# -------------------------- 可修改参数 --------------------------
STRENGTH = 0.5         # 编辑强度(0~1)：越大改动越大
# -----------------------------------------------------------------

def _load_source_image():
    """优先用上传的源图；未上传则生成程序化测试图"""
    if test_data_path and os.path.isfile(test_data_path):
        img = cv2.imread(test_data_path)
        assert img is not None, f"无法读取图片：{test_data_path}"
        print(f"使用上传的源图: {test_data_path}")
    else:
        print("未上传源图，使用程序化测试图（彩色渐变+几何图形）演示")
        h = w = getattr(model, 'image_size', 32)
        img = np.zeros((h, w, 3), np.uint8)
        for y in range(h):
            img[y, :] = (int(255 * y / h), 128, int(255 - 255 * y / h))
        cv2.circle(img, (w // 2, h // 2), max(h // 4, 2), (255, 255, 255), -1)
    return img

# 1. 加载模型
print("正在加载扩散编辑模型...")
model = load_model(model_path, map_location='cpu')
model.eval()
print(f"✅ 加载成功，参数量: {sum(p.numel() for p in model.parameters()):,}")

if STRENGTH < model.num_timesteps:
    pass
src_bgr = _load_source_image()
size = model.image_size
src_bgr = cv2.resize(src_bgr, (size, size), interpolation=cv2.INTER_AREA)
src_rgb = cv2.cvtColor(src_bgr, cv2.COLOR_BGR2RGB)

# 2. 预处理到 [-1,1] 并执行编辑
x = torch.from_numpy(src_rgb.astype(np.float32) / 255.0).permute(2, 0, 1).unsqueeze(0) * 2 - 1
steps = min(STRENGTH if isinstance(STRENGTH, int) else int(model.num_timesteps * float(STRENGTH)),
            model.num_timesteps)
if steps < model.num_timesteps:
    model.num_timesteps = steps
    print(f"⚠️ 本次编辑强度对应 {steps} 步")
print(f"开始编辑（强度 {STRENGTH}）...")
out = model.edit(x, strength=float(min(STRENGTH, 0.99)))

# 3. 保存结果并计算重建误差
out_arr = ((out[0].clamp(-1, 1) + 1) / 2 * 255).byte().permute(1, 2, 0).numpy()
out_dir = os.path.dirname(os.path.abspath(model_path))
result_p = os.path.join(out_dir, 'edited_result.png')
cv2.imwrite(result_p, cv2.cvtColor(out_arr, cv2.COLOR_RGB2BGR))

mse = float(np.mean((out_arr.astype(np.float32) - src_rgb.astype(np.float32)) ** 2))
print(f"✅ 编辑完成，结果已保存: {result_p}")
print(f"与源图的像素 MSE: {mse:.2f}（越小越接近原图）")
print(f"损失：{min(mse, 1e6):.4f}")
print("=== 测试完成 ===")
