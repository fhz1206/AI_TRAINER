# ViT 图像分类模型测试模板
import torch
import os
from torch.utils.data import DataLoader
from model import ImageDataset
from model_io import load_model

# -------------------------- 可修改参数 --------------------------
BATCH_SIZE = 32
# -----------------------------------------------------------------

# 1. 加载模型（自动注入模型路径；元数据自动重建 ViT 结构与注意力积木）
print("正在加载 ViT 模型...")
model = load_model(model_path, map_location='cpu')
model.eval()
meta = getattr(model, '_metadata', {})
print(f"✅ 模型加载成功，参数量: {sum(p.numel() for p in model.parameters()):,}")
print(f"架构: {meta.get('architecture', 'VisionTransformer')} | "
      f"注意力: {meta.get('model_params', {}).get('attention_type', '?')}")

# 2. 加载测试数据集（自动注入测试数据路径）
dataset = ImageDataset(test_data_path,
                       image_size=meta.get('model_params', {}).get('image_size', 224))
loader = DataLoader(dataset, batch_size=BATCH_SIZE, shuffle=False)
print(f"测试集共 {len(dataset)} 个样本，{dataset.num_classes} 个类别")
class_names = list(dataset.classes.keys())

# 3. 逐批评估
correct = 0
total = 0
with torch.no_grad():
    for batch_idx, (imgs, labels) in enumerate(loader):
        outputs = model(imgs)
        preds = outputs.argmax(1)
        correct += preds.eq(labels).sum().item()
        total += labels.size(0)
        print(f"批次 {batch_idx+1}/{len(loader)} | 累计正确 {correct}/{total}")

accuracy = 100. * correct / max(total, 1)
print("\n" + "=" * 60)
print(f"总样本数: {total} | 正确数: {correct}")
# 保留固定格式，方便后端解析指标
print(f"准确率：{accuracy:.2f}%")
